function sendAfter_(sh, rows, i, pages, now) {
  const r = rows[i], line = i + 1, t = pubTime_(r), v = String(r[C.VALIDATION]);
  const pg = pageFor_(pages, r);
  if (!pg || now - t < 5 * 60000) return;
  if (!pg.apres || now - t > 3 * 3600000) { mark_(sh, line, v + ' | Confirmation non envoyée'); return; }
  const nom = String(r[C.PAGE] || '').trim();
  const next = [], tasks = [];
  for (let k = 1; k < rows.length; k++) {
    const x = rows[k];
    if (!x[C.ID] || k === i) continue;
    const st = String(x[C.STATUT]);
    if (String(x[C.RESEAU]) === 'Tâche') { if (st === 'À faire' && tasks.length < 4) tasks.push(x); continue; }
    if (String(x[C.PAGE] || '').trim() !== nom || st === 'Publié' || st === 'Refusé' || st === 'À publier à la main') continue;
    const tx = pubTime_(x);
    if (tx > now) next.push({ t: tx, x: x });
  }
  next.sort(function (a, b) { return a.t - b.t; });
  const li = function (arr) { return arr.length ? '<ul>' + arr.join('') + '</ul>' : '<p style="color:#4a5b73">Rien pour le moment.</p>'; };
  const nextHtml = li(next.slice(0, 5).map(function (n) { return '<li>' + esc_(heureFr_(n.t)) + ' : ' + esc_(n.x[C.SUJET]) + ' (' + esc_(n.x[C.ID]) + ')</li>'; }));
  const taskHtml = li(tasks.map(function (x) { return '<li>' + esc_(heureFr_(pubTime_(x))) + ' : ' + esc_(x[C.SUJET]) + '</li>'; }));
  const html = '<div style="font-family:Arial,sans-serif;max-width:620px;margin:auto;color:#0b1f3a">' +
    '<div style="background:#0b1f3a;color:#ffffff;padding:18px 22px;border-radius:10px 10px 0 0"><strong>Publié</strong><br>' + esc_(pg.nom) + ' · ' + esc_(r[C.ID]) + ' · ' + esc_(heureFr_(t)) + '</div>' +
    '<div style="border:1px solid #d6dde8;border-top:none;padding:20px 22px;border-radius:0 0 10px 10px">' +
    '<p><b>' + esc_(r[C.SUJET]) + '</b> est parti sur <b>' + esc_(pg.nom) + '</b>.</p><p style="color:#4a5b73;font-size:13px">' + esc_(v) + '</p>' +
    '<h3 style="margin-bottom:4px">Prochaines publications</h3>' + nextHtml +
    '<h3 style="margin-bottom:4px">Tâches à faire</h3>' + taskHtml +
    '</div></div>';
  const o = { to: pg.email || adminMail_(), subject: 'Publié : ' + pg.nom + ' ' + r[C.ID] + ' à ' + Utilities.formatDate(t, tz_(), 'HH:mm'), htmlBody: html, name: 'Agent social media' };
  if (pg.copie) o.cc = pg.copie;
  MailApp.sendEmail(o);
  mark_(sh, line, v + ' | Confirmation envoyée le ' + stamp_(now));
}

function igUser_(pg, tok) {
  const res = UrlFetchApp.fetch('https://graph.facebook.com/' + pg.id + '?fields=instagram_business_account&access_token=' + encodeURIComponent(tok), { muteHttpExceptions: true });
  const j = JSON.parse(res.getContentText());
  if (!j.instagram_business_account) throw new Error('Compte Instagram professionnel non relié à cette Page, ou droits Instagram manquants : ' + res.getContentText().slice(0, 200));
  return j.instagram_business_account.id;
}

function igCall_(path, payload) {
  const res = UrlFetchApp.fetch('https://graph.facebook.com/' + path, { method: 'post', muteHttpExceptions: true, payload: payload });
  const j = JSON.parse(res.getContentText());
  if (!j.id) throw new Error('Instagram : ' + res.getContentText().slice(0, 300));
  return j.id;
}

function igWait_(id, tok) {
  for (let k = 0; k < 8; k++) {
    const j = JSON.parse(UrlFetchApp.fetch('https://graph.facebook.com/' + id + '?fields=status_code&access_token=' + encodeURIComponent(tok), { muteHttpExceptions: true }).getContentText());
    if (j.status_code === 'FINISHED') return;
    if (j.status_code === 'ERROR') throw new Error('Instagram a refusé l\'image.');
    Utilities.sleep(3000);
  }
}

function postInstagram_(pg, caption, visuel) {
  const tok = fbToken_(pg), ig = igUser_(pg, tok), urls = urls_(visuel).slice(0, 10);
  if (!urls.length) throw new Error('Instagram exige une image.');
  let creation;
  if (urls.length === 1) creation = igCall_(ig + '/media', { image_url: urls[0], caption: caption, access_token: tok });
  else {
    const kids = urls.map(function (u) { const c = igCall_(ig + '/media', { image_url: u, is_carousel_item: 'true', access_token: tok }); igWait_(c, tok); return c; });
    creation = igCall_(ig + '/media', { media_type: 'CAROUSEL', children: kids.join(','), caption: caption, access_token: tok });
  }
  igWait_(creation, tok);
  return igCall_(ig + '/media_publish', { creation_id: creation, access_token: tok });
}
