const SH_CAL = 'Calendrier', SH_PAGES = 'Pages', SH_HELP = 'Mode emploi';
const C = { ID: 0, DATE: 1, HEURE: 2, RESEAU: 3, PILIER: 4, SUJET: 5, TEXTE: 7, VISUEL: 9, MODE: 11, STATUT: 12, VALIDATION: 13, PAGE: 14 };
const HEADERS = ['ID', 'Date', 'Heure', 'Reseau', 'Pilier', 'Sujet', 'Accroche', 'Texte', 'Appel a action', 'Visuel', 'Sources', 'Mode', 'Statut', 'Validation', 'Page'];
const PHEAD = ['Nom', 'Actif', 'Reseau', 'Identifiant Page Facebook', 'Email de validation', 'Email en copie', 'Regle posts rediges', 'Pas de mail la nuit', 'Veille actu', 'Themes veille', 'Dossier affiches', 'Remarque'];
const LEAD_MIN = 30, GRACE_MIN = 120;
const LI_CLIENT_ID_DEFAULT = '78qp5jvajpikuv';
const RAW_DEFAULT = 'https://raw.githubusercontent.com/gnonlonfounwilfried9-dotcom/wilactu-affiches/main/';

// ---------- outils ----------
function props_() { return PropertiesService.getScriptProperties(); }
function tz_() { return props_().getProperty('TZ') || 'Africa/Porto-Novo'; }
function noAcc_(s) { return String(s || '').normalize('NFD').replace(/[̀-ͯ]/g, ''); }
function key_(s) { return noAcc_(s).toLowerCase().replace(/[^a-z0-9]+/g, '_').replace(/^_|_$/g, ''); }
function fmt_(v, pattern) { return v instanceof Date ? Utilities.formatDate(v, tz_(), pattern) : String(v).trim(); }
function pubTime_(r) { return Utilities.parseDate(fmt_(r[C.DATE], 'yyyy-MM-dd') + ' ' + fmt_(r[C.HEURE], 'HH:mm'), tz_(), 'yyyy-MM-dd HH:mm'); }
function stamp_(d) { return Utilities.formatDate(d, tz_(), 'yyyy-MM-dd HH:mm'); }
function mark_(sh, line, text) { sh.getRange(line, C.VALIDATION + 1).setValue(text); }
function cal_() { return SpreadsheetApp.getActive().getSheetByName(SH_CAL); }
function esc_(s) { return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }
function webappUrl_() { return props_().getProperty('WEBAPP_URL') || ScriptApp.getService().getUrl(); }
function adminMail_() { return props_().getProperty('ADMIN_EMAIL') || Session.getEffectiveUser().getEmail(); }

function heureFr_(t) {
  const jours = ['dimanche', 'lundi', 'mardi', 'mercredi', 'jeudi', 'vendredi', 'samedi'];
  const mois = ['janvier', 'février', 'mars', 'avril', 'mai', 'juin', 'juillet', 'août', 'septembre', 'octobre', 'novembre', 'décembre'];
  const j = Number(Utilities.formatDate(t, tz_(), 'u')) % 7;
  const n = Number(Utilities.formatDate(t, tz_(), 'M')) - 1;
  return jours[j] + ' ' + Utilities.formatDate(t, tz_(), 'd') + ' ' + mois[n] + ' à ' + Utilities.formatDate(t, tz_(), 'HH:mm');
}

function sign_(value) {
  const bytes = Utilities.computeHmacSha256Signature(value, props_().getProperty('SECRET'));
  return bytes.map(function (b) { return ('0' + (b & 255).toString(16)).slice(-2); }).join('').slice(0, 24);
}

function page_(message) {
  return HtmlService.createHtmlOutput('<div style="font-family:Arial,sans-serif;max-width:480px;margin:60px auto;padding:24px;border:1px solid #d6dde8;border-radius:10px;color:#0b1f3a;font-size:18px">' + esc_(message) + '</div>');
}

function say_(title, text) {
  try { const ui = SpreadsheetApp.getUi(); ui.alert(title, text, ui.ButtonSet.OK); }
  catch (e) { alert_(title, text, null); }
}

function alert_(subject, body, pg) {
  const o = { to: pg && pg.email ? pg.email : adminMail_(), subject: 'Agent social media : ' + subject, body: body, name: 'Agent social media' };
  const cc = pg && pg.copie ? pg.copie : '';
  if (cc) o.cc = cc;
  MailApp.sendEmail(o);
}

// ---------- pages ----------
function pages_() {
  const sh = SpreadsheetApp.getActive().getSheetByName(SH_PAGES);
  if (!sh || sh.getLastRow() < 2) return [];
  return sh.getRange(2, 1, sh.getLastRow() - 1, PHEAD.length).getValues().map(function (v, i) {
    const low = function (x) { return String(x).trim().toLowerCase(); };
    return {
      row: i + 2, nom: String(v[0]).trim(), key: key_(v[0]), actif: low(v[1]) === 'oui', reseau: String(v[2]).trim(),
      id: String(v[3]).trim(), email: String(v[4]).trim(), copie: String(v[5]).trim(), regle: low(v[6]) || 'veto',
      nuit: low(v[7]) === 'oui', veille: low(v[8]) === 'oui', themes: low(v[9]).split(/[,; ]+/).filter(Boolean),
      affiches: String(v[10]).trim()
    };
  }).filter(function (p) { return p.nom; });
}

function pageFor_(pages, r) {
  const nom = String(r[C.PAGE] || '').trim();
  const net = String(r[C.RESEAU]).trim();
  const ok = pages.filter(function (p) { return p.actif && p.reseau === net; });
  if (nom) return ok.filter(function (p) { return p.nom === nom; })[0] || null;
  return ok.length === 1 ? ok[0] : null;
}

function selectedPage_() {
  const sh = SpreadsheetApp.getActiveSheet();
  if (sh.getName() !== SH_PAGES) return null;
  const row = sh.getActiveRange().getRow();
  return pages_().filter(function (p) { return p.row === row; })[0] || null;
}

// ---------- boucle principale (toutes les 5 minutes) ----------
function tick() {
  const sh = cal_(), rows = sh.getDataRange().getValues(), now = new Date(), pages = pages_();
  for (let i = 1; i < rows.length; i++) {
    const r = rows[i];
    if (!r[C.ID]) continue;
    const line = i + 1, statut = String(r[C.STATUT]);
    let validation = String(r[C.VALIDATION]);
    if (statut === 'Publié' || statut === 'Refusé' || validation.indexOf('Terminé') === 0) continue;
    const t = pubTime_(r), minutesToGo = (t - now) / 60000;
    if (minutesToGo <= 180 && minutesToGo > -GRACE_MIN && r[C.MODE] === 'Jour J' && !String(r[C.TEXTE]).trim()) {
      const g = genJourJ_(r);
      if (g) { sh.getRange(line, C.TEXTE + 1).setValue(g); r[C.TEXTE] = g; }
    }
    if (minutesToGo > LEAD_MIN) continue;
    const pg = pageFor_(pages, r);
    const pending = validation.indexOf('Texte manquant') === 0 || validation.indexOf('Page introuvable') === 0;
    if (!pg) {
      if (!pending) {
        mark_(sh, line, 'Page introuvable, alerte envoyée');
        alert_('Page introuvable pour ' + r[C.ID], 'La ligne ' + r[C.ID] + ' ne correspond à aucune page active. Remplis la colonne Page avec le nom exact d\'une page active de l\'onglet Pages, et vérifie le réseau.', null);
      }
      continue;
    }
    const texte = String(r[C.TEXTE]).trim();
    if (!texte) {
      if (!pending) {
        mark_(sh, line, 'Texte manquant, alerte envoyée');
        alert_('Texte manquant pour ' + r[C.ID], 'La ligne ' + r[C.ID] + ' (' + r[C.SUJET] + ') est prévue dans moins de ' + LEAD_MIN + ' minutes mais son texte est vide.', pg);
      }
      continue;
    }
    if (pending) validation = '';
    if (!validation) {
      sh.getRange(line, C.STATUT + 1).setValue('À valider');
      if (silent_(r, pg, t)) { mark_(sh, line, 'Nuit : publication directe, sans mail'); continue; }
      sendApproval_(r, t, pg);
      mark_(sh, line, 'Mail envoyé le ' + stamp_(now));
      continue;
    }
    if (minutesToGo <= 0) handleDue_(sh, r, line, minutesToGo, pg);
  }
}

function night_(t) { const h = Number(Utilities.formatDate(t, tz_(), 'H')); return h >= 22 || h < 7; }
function silent_(r, pg, t) { return String(r[C.MODE]) === 'Rédigé' && pg.nuit && pg.regle !== 'validation' && night_(t); }

function handleDue_(sh, r, line, minutesToGo, pg) {
  const statut = String(r[C.STATUT]), mode = String(r[C.MODE]);
  if (minutesToGo < -GRACE_MIN) { mark_(sh, line, 'Terminé : hors délai, non publié'); return; }
  const autoOk = statut === 'À valider' && (mode === 'Alerte' || (mode === 'Rédigé' && pg.regle !== 'validation'));
  if (statut === 'Validé' || autoOk) publishRow_(sh, r, line, pg);
  else if (statut === 'À valider') mark_(sh, line, 'Terminé : sans réponse, non publié');
}

function publishRow_(sh, r, line, pg) {
  if (props_().getProperty('PUBLISH_ENABLED') !== 'true') {
    mark_(sh, line, 'Terminé : simulation, aurait publié sur ' + pg.nom);
    return;
  }
  try {
    const id = pg.reseau === 'Facebook' ? postFacebook_(pg, String(r[C.TEXTE]), String(r[C.VISUEL])) : postLinkedIn_(pg, String(r[C.TEXTE]), String(r[C.VISUEL]));
    sh.getRange(line, C.STATUT + 1).setValue('Publié');
    mark_(sh, line, 'Terminé : publié le ' + stamp_(new Date()) + ' (' + id + ')');
  } catch (e) {
    mark_(sh, line, 'Erreur : ' + e.message);
    alert_('Erreur de publication ' + r[C.ID] + ' (' + pg.nom + ')', e.message, pg);
  }
}

// ---------- mail de validation ----------
function sendApproval_(r, t, pg) {
  const base = webappUrl_(), id = String(r[C.ID]), mode = String(r[C.MODE]);
  const link = function (a) { return base + '?a=' + a + '&id=' + encodeURIComponent(id) + '&s=' + sign_(id + a); };
  const regle = mode === 'Alerte'
    ? 'Alerte actu : ce post sera publié automatiquement à l\'heure indiquée, dans environ 10 minutes. Clique sur Refuser pour l\'annuler.'
    : (mode === 'Rédigé' && pg.regle !== 'validation')
      ? 'Sans réponse de ta part, ce post rédigé à l\'avance sera publié à l\'heure prévue. Clique sur Refuser pour le bloquer.'
      : 'Sans réponse de ta part, ce post ne sera pas publié.';
  const btn = function (href, label, bg) {
    return '<a href="' + href + '" style="display:inline-block;padding:12px 22px;margin-right:10px;border-radius:8px;background:' + bg + ';color:#ffffff;text-decoration:none;font-weight:600">' + label + '</a>';
  };
  const html = '<div style="font-family:Arial,sans-serif;max-width:620px;margin:auto;color:#0b1f3a">' +
    '<div style="background:#0b1f3a;color:#ffffff;padding:18px 22px;border-radius:10px 10px 0 0"><strong>' + (mode === 'Alerte' ? 'Alerte actu' : 'Validation requise') + '</strong><br>' + esc_(pg.nom) + ' · ' + esc_(pg.reseau) + ' · ' + esc_(id) + ' · ' + esc_(heureFr_(t)) + '</div>' +
    '<div style="border:1px solid #d6dde8;border-top:none;padding:20px 22px;border-radius:0 0 10px 10px">' +
    '<p style="color:#4a5b73;margin-top:0">Sujet : ' + esc_(r[C.SUJET]) + '</p>' +
    (/^https?:\/\//.test(String(r[C.VISUEL])) ? '<p><img src="' + esc_(r[C.VISUEL]) + '" style="max-width:100%;border-radius:8px"></p>' : '') +
    '<div style="background:#f3f6fb;border-radius:8px;padding:16px;line-height:1.55;white-space:pre-wrap">' + esc_(r[C.TEXTE]) + '</div>' +
    '<p style="margin:20px 0">' + btn(link('ok'), 'Valider', '#1f5fbf') + btn(link('no'), 'Refuser', '#6b7a90') + '</p>' +
    '<p style="color:#4a5b73;font-size:13px">' + esc_(regle) + '</p>' +
    '<p style="color:#4a5b73;font-size:13px">Pour modifier le texte, change le dans le calendrier avant l\'heure de publication, puis clique sur Valider.</p>' +
    '</div></div>';
  const o = {
    to: pg.email || adminMail_(),
    subject: (mode === 'Alerte' ? 'Alerte actu, publication dans 10 minutes : ' : 'Validation requise : ') + pg.nom + ' ' + id + ' à ' + Utilities.formatDate(t, tz_(), 'HH:mm'),
    htmlBody: html, name: 'Agent social media'
  };
  if (pg.copie) o.cc = pg.copie;
  MailApp.sendEmail(o);
}

// ---------- lien cliqué depuis un mail ----------
function doGet(e) {
  if (e.parameter.code) return handleLinkedInCode_(e.parameter.code, e.parameter.state);
  const a = e.parameter.a, id = e.parameter.id, s = e.parameter.s;
  if ((a !== 'ok' && a !== 'no') || !id || !s || s !== sign_(id + a)) return page_('Lien invalide.');
  const sh = cal_(), rows = sh.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][C.ID]) !== id) continue;
    const line = i + 1;
    if (String(rows[i][C.STATUT]) === 'Publié') return page_('Ce post est déjà publié. Rien à changer.');
    if (String(rows[i][C.VALIDATION]).indexOf('Terminé') === 0) return page_('Trop tard : l\'heure de publication est passée.');
    sh.getRange(line, C.STATUT + 1).setValue(a === 'ok' ? 'Validé' : 'Refusé');
    mark_(sh, line, 'Réponse reçue (' + (a === 'ok' ? 'validé' : 'refusé') + ') le ' + stamp_(new Date()));
    return page_(a === 'ok' ? 'Post validé. Il partira à l\'heure prévue.' : 'Post refusé. Il ne sera pas publié.');
  }
  return page_('Post introuvable.');
}

// ---------- Facebook ----------
function fbToken_(pg) {
  const p = props_(), sys = p.getProperty('FB_TOKEN_' + pg.key) || p.getProperty('FB_SYSTEM_TOKEN');
  if (!sys) throw new Error('Clé Facebook manquante : renseigne FB_SYSTEM_TOKEN (ou FB_TOKEN_' + pg.key + ') dans les propriétés du script.');
  if (!pg.id) throw new Error('Identifiant de Page Facebook manquant pour ' + pg.nom);
  const res = UrlFetchApp.fetch('https://graph.facebook.com/' + pg.id + '?fields=access_token&access_token=' + encodeURIComponent(sys), { muteHttpExceptions: true });
  const j = JSON.parse(res.getContentText());
  if (!j.access_token) throw new Error('Facebook a refusé l\'accès à la Page ' + pg.nom + ' : ' + res.getContentText().slice(0, 300));
  return j.access_token;
}

function postFacebook_(pg, message, visuel) {
  const tok = fbToken_(pg);
  if (/^https?:\/\//.test(String(visuel || '').trim())) {
    const r1 = UrlFetchApp.fetch('https://graph.facebook.com/' + pg.id + '/photos', {
      method: 'post', muteHttpExceptions: true,
      payload: { url: String(visuel).trim(), caption: message, published: 'true', access_token: tok }
    });
    const j1 = JSON.parse(r1.getContentText());
    if (j1.post_id || j1.id) return j1.post_id || j1.id;
  }
  const res = UrlFetchApp.fetch('https://graph.facebook.com/' + pg.id + '/feed', {
    method: 'post', muteHttpExceptions: true, payload: { message: message, access_token: tok }
  });
  const j = JSON.parse(res.getContentText());
  if (!j.id) throw new Error(res.getContentText());
  return j.id;
}

// ---------- LinkedIn ----------
function liClientId_() { return props_().getProperty('LI_CLIENT_ID') || LI_CLIENT_ID_DEFAULT; }

function liAuthUrl_(pg) {
  return 'https://www.linkedin.com/oauth/v2/authorization?response_type=code&client_id=' + liClientId_() +
    '&redirect_uri=' + encodeURIComponent(webappUrl_()) + '&state=' + encodeURIComponent(pg.key + '.' + sign_('li' + pg.key)) +
    '&scope=' + encodeURIComponent('openid profile w_member_social');
}

function handleLinkedInCode_(code, state) {
  const parts = String(state || '').split('.');
  if (parts.length !== 2 || parts[1] !== sign_('li' + parts[0])) return page_('Lien invalide.');
  const pg = pages_().filter(function (p) { return p.key === parts[0]; })[0];
  if (!pg) return page_('Page introuvable dans l\'onglet Pages.');
  const p = props_(), secret = p.getProperty('LI_CLIENT_SECRET');
  if (!secret) return page_('Le secret de l\'application LinkedIn (LI_CLIENT_SECRET) manque dans les propriétés du script.');
  const res = UrlFetchApp.fetch('https://www.linkedin.com/oauth/v2/accessToken', {
    method: 'post', muteHttpExceptions: true,
    payload: { grant_type: 'authorization_code', code: code, redirect_uri: webappUrl_(), client_id: liClientId_(), client_secret: secret }
  });
  const j = JSON.parse(res.getContentText());
  if (!j.access_token) return page_('Échec LinkedIn : ' + (j.error_description || res.getContentText().slice(0, 200)));
  const ui = JSON.parse(UrlFetchApp.fetch('https://api.linkedin.com/v2/userinfo', { headers: { Authorization: 'Bearer ' + j.access_token }, muteHttpExceptions: true }).getContentText());
  if (!ui.sub) return page_('LinkedIn n\'a pas renvoyé l\'identité du profil.');
  const exp = Utilities.formatDate(new Date(Date.now() + Number(j.expires_in || 5184000) * 1000), tz_(), 'yyyy-MM-dd');
  p.setProperty('LI_TOKEN_' + pg.key, j.access_token);
  p.setProperty('LI_URN_' + pg.key, 'urn:li:person:' + ui.sub);
  p.setProperty('LI_EXP_' + pg.key, exp);
  alert_('LinkedIn connecté pour ' + pg.nom, 'Le profil ' + (ui.name || '') + ' est connecté. Le jeton expire le ' + exp + '. Un rappel sera envoyé 7 jours avant.', pg);
  return page_('LinkedIn est connecté pour ' + pg.nom + '. Tu peux fermer cette page.');
}

function postLinkedIn_(pg, text, visuel) {
  const p = props_(), tok = p.getProperty('LI_TOKEN_' + pg.key), author = p.getProperty('LI_URN_' + pg.key);
  if (!tok || !author) throw new Error('LinkedIn n\'est pas connecté pour ' + pg.nom + '. Utilise le menu Agent social media, Envoyer le lien LinkedIn.');
  const hdr = { Authorization: 'Bearer ' + tok, 'X-Restli-Protocol-Version': '2.0.0' };
  let media = null;
  if (/^https?:\/\//.test(String(visuel || '').trim())) { try { media = liUploadImage_(String(visuel).trim(), author, hdr); } catch (e) { media = null; } }
  const content = media
    ? { shareCommentary: { text: text }, shareMediaCategory: 'IMAGE', media: [{ status: 'READY', media: media }] }
    : { shareCommentary: { text: text }, shareMediaCategory: 'NONE' };
  const res = UrlFetchApp.fetch('https://api.linkedin.com/v2/ugcPosts', {
    method: 'post', contentType: 'application/json', headers: hdr, muteHttpExceptions: true,
    payload: JSON.stringify({ author: author, lifecycleState: 'PUBLISHED', specificContent: { 'com.linkedin.ugc.ShareContent': content }, visibility: { 'com.linkedin.ugc.MemberNetworkVisibility': 'PUBLIC' } })
  });
  const j = JSON.parse(res.getContentText());
  if (!j.id) throw new Error(res.getContentText().slice(0, 300));
  return j.id;
}

function liUploadImage_(url, author, hdr) {
  const reg = UrlFetchApp.fetch('https://api.linkedin.com/v2/assets?action=registerUpload', {
    method: 'post', contentType: 'application/json', headers: hdr, muteHttpExceptions: true,
    payload: JSON.stringify({ registerUploadRequest: { recipes: ['urn:li:digitalmediaRecipe:feedshare-image'], owner: author, serviceRelationships: [{ relationshipType: 'OWNER', identifier: 'urn:li:userGeneratedContent' }] } })
  });
  const j = JSON.parse(reg.getContentText());
  const up = j.value.uploadMechanism['com.linkedin.digitalmedia.uploading.MediaUploadHttpRequest'].uploadUrl;
  const blob = UrlFetchApp.fetch(url).getBlob();
  UrlFetchApp.fetch(up, { method: 'put', headers: { Authorization: hdr.Authorization }, payload: blob.getBytes(), contentType: blob.getContentType(), muteHttpExceptions: true });
  return j.value.asset;
}

function tokenReminder() {
  pages_().forEach(function (pg) {
    if (!pg.actif || pg.reseau !== 'LinkedIn') return;
    const exp = props_().getProperty('LI_EXP_' + pg.key);
    if (!exp) return;
    const days = (Utilities.parseDate(exp, tz_(), 'yyyy-MM-dd') - new Date()) / 86400000;
    if (days <= 7 && days > -1) alert_('Jeton LinkedIn bientôt expiré pour ' + pg.nom, 'Le jeton expire le ' + exp + '. Dans le menu Agent social media, choisis Envoyer le lien LinkedIn, puis clique sur Autoriser.', pg);
  });
}

function strip_(s) {
  return String(s || '').replace(/<[^>]*>/g, ' ').replace(/&nbsp;/g, ' ').replace(/&amp;/g, '&').replace(/&quot;/g, '"').replace(/&#8217;|&#039;|&apos;/g, "'").replace(/&#8230;/g, '...').replace(/&#\d+;/g, ' ').replace(/\s+/g, ' ').trim();
}
function noDash_(s) {
  return String(s).replace(/[‒-―−]/g, ',').replace(/\s-\s/g, ', ').replace(/[\u{1F000}-\u{1FAFF}\u{2600}-\u{27BF}\u{FE0F}]/gu, '').replace(/\s+,/g, ',').replace(/\s+/g, ' ').trim();
}
function noDashBlock_(s) { return String(s).split('\n').map(noDash_).join('\n').replace(/\n{3,}/g, '\n\n').trim(); }
function cut_(s, n) {
  if (s.length <= n) return s;
  const c = s.slice(0, n), k = Math.max(c.lastIndexOf('. '), c.lastIndexOf(' '));
  let o = c.slice(0, k > 80 ? k : n).replace(/[,;:]$/, '');
  const q = o.lastIndexOf('«');
  if (q > 40 && q > o.lastIndexOf('»')) o = o.slice(0, q).trim();
  return o + '...';
}

function feedItems_(f) {
  const res = UrlFetchApp.fetch(f.url, { muteHttpExceptions: true, followRedirects: true, headers: { 'User-Agent': 'Mozilla/5.0 (compatible; AgentSocialMedia)' } });
  if (res.getResponseCode() !== 200) return [];
  const ch = XmlService.parse(res.getContentText()).getRootElement().getChild('channel');
  const out = [];
  if (!ch) return out;
  const txt = function (el, n) { const c = el.getChild(n); return c ? c.getText() : ''; };
  ch.getChildren('item').forEach(function (it) {
    const d = new Date(txt(it, 'pubDate'));
    if (isNaN(d.getTime())) return;
    out.push({ src: f.src, lang: f.lang, cat: f.cat, title: strip_(txt(it, 'title')), link: txt(it, 'link').trim(), desc: strip_(txt(it, 'description')), date: d });
  });
  return out;
}

function annotate_(i) {
  const words = noAcc_(i.title).toLowerCase().replace(/[^a-z0-9 ]/g, ' ').split(/\s+/).filter(Boolean);
  const ev = [], ent = [];
  words.forEach(function (w) {
    Object.keys(CLASSES).forEach(function (k) { if (CLASSES[k].indexOf(w) >= 0 && ev.indexOf(k) < 0) ev.push(k); });
    if (ENTITIES.indexOf(w) >= 0 && ent.indexOf(w) < 0) ent.push(w);
  });
  i.ev = ev; i.ent = ent;
  return i;
}

function shares_(a, b) {
  return a.ev.some(function (e) { return b.ev.indexOf(e) >= 0; }) && a.ent.some(function (e) { return b.ent.indexOf(e) >= 0; });
}

function catOf_(ent, fallback) {
  const iC = ENTITIES.indexOf('mbappe'), iT = ENTITIES.indexOf('openai');
  const idx = ent.map(function (e) { return ENTITIES.indexOf(e); }).sort(function (a, b) { return a - b; })[0];
  if (idx === undefined) return fallback;
  return idx < iC ? 'crypto' : (idx < iT ? 'sport' : 'tech');
}

function gemini_(prompt) {
  const key = props_().getProperty('GEMINI_API_KEY');
  if (!key) return null;
  try {
    const res = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/gemini-flash-latest:generateContent', {
      method: 'post', contentType: 'application/json', muteHttpExceptions: true, headers: { 'x-goog-api-key': key },
      payload: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
    });
    return JSON.parse(res.getContentText()).candidates[0].content.parts[0].text;
  } catch (e) { return null; }
}

function tagOf_(pg) { return '#' + pg.nom.replace(/[^A-Za-z0-9]/g, ''); }

function brkText_(pg, cat, main, group) {
  const names = [main.src];
  group.forEach(function (g) { if (names.indexOf(g.src) < 0) names.push(g.src); });
  const srcs = names.length > 1 ? names.slice(0, -1).join(', ') + ' et ' + names[names.length - 1] : names[0];
  const title = noDash_(main.title);
  const chapo = cut_(noDash_(main.desc.replace(/(L'article|Cet article|The post|Le post).{0,200}(apparu en premier sur|appeared first on).*$/i, '')), 230);
  const g = gemini_('Tu es le rédacteur de la Page Facebook ' + pg.nom + ' (' + pg.themes.join(', ') + '). Réécris cette information urgente en français, en 3 à 4 phrases courtes, ton vivant et clair, sans emoji, sans tiret, sans inventer aucun fait absent des extraits. Commence par ' + LABELS[cat] + ' sur sa propre ligne. Termine par une question qui invite à commenter, puis cite les sources : ' + srcs + '. Extraits : ' + [main].concat(group).map(function (x) { return x.title + '. ' + x.desc.slice(0, 300); }).join(' | '));
  if (g) return noDashBlock_(g) + '\n\n' + tagOf_(pg) + ' ' + TAGS[cat];
  return LABELS[cat] + '\n\n' + title + '\n\n' + (chapo && chapo !== title ? chapo + '\n\n' : '') + 'Source : ' + srcs + '.\n\nVotre avis ? Dites en commentaire ce que vous en pensez.\n\n' + tagOf_(pg) + ' ' + TAGS[cat];
}

// ---------- agent d'actualité (toutes les 10 minutes) ----------
function breaking_() { runBreaking_(false, 0); }
function testBreaking() { runBreaking_(true, 72); }

function runBreaking_(dry, wide) {
  const p = props_();
  if (p.getProperty('BRK_ENABLED') === 'false') return;
  const targets = pages_().filter(function (g) { return g.actif && g.veille && g.reseau === 'Facebook'; });
  if (!targets.length) { if (dry) say_('Veille actu', 'Aucune page active avec Veille actu = oui.'); return; }
  const now = new Date();
  let items = [], feedsOk = 0;
  FEEDS.forEach(function (f) { try { items = items.concat(feedItems_(f)); feedsOk++; } catch (e) {} });
  const recent = items.filter(function (i) { return now - i.date < (wide ? wide * 3600000 : 4 * 3600000) && now - i.date > -600000; }).map(annotate_);
  const cands = recent.filter(function (i) { return i.ev.length && i.ent.length && now - i.date < (wide ? wide * 3600000 : 120 * 60000); });
  const stories = cands.map(function (c) {
    const group = recent.filter(function (o) { return o.src !== c.src && shares_(c, o); });
    const srcCount = 1 + group.map(function (g) { return g.src; }).filter(function (s, k, a) { return a.indexOf(s) === k; }).length;
    const fr = [c].concat(group).filter(function (x) { return x.lang === 'fr'; }).sort(function (a, b) { return a.date - b.date; })[0];
    return { c: c, group: group, srcCount: srcCount, main: fr || null, cat: catOf_(c.ent, fr ? fr.cat : c.cat), sig: c.ev[0] + '|' + c.ent.slice().sort()[0] };
  }).filter(function (s) { return s.srcCount >= 2 && (s.main || p.getProperty('GEMINI_API_KEY')); });
  let report = 'Flux lus : ' + feedsOk + ' sur ' + FEEDS.length + '. Articles récents : ' + recent.length + '. Candidats : ' + cands.length + '. Confirmés par 2 sources ou plus : ' + stories.length + '.\n';
  targets.forEach(function (pg) {
    const log = JSON.parse(p.getProperty('BRK_LOG_' + pg.key) || '[]').filter(function (t) { return now - t < 24 * 3600000; });
    const seen = JSON.parse(p.getProperty('BRK_SEEN_' + pg.key) || '[]').filter(function (x) { return now - x.t < 12 * 3600000; });
    const good = stories.filter(function (s) { return (!pg.themes.length || pg.themes.indexOf(s.cat) >= 0) && !seen.some(function (x) { return x.sig === s.sig; }); })
      .sort(function (a, b) { return b.srcCount - a.srcCount || b.c.date - a.c.date; });
    report += '\n' + pg.nom + ' : ' + good.length + ' sujet(s) possible(s).\n';
    if (!good.length) return;
    const s = good[0], main = s.main || s.c, text = brkText_(pg, s.cat, main, s.group);
    const base = pg.affiches ? (pg.affiches.slice(-1) === '/' ? pg.affiches : pg.affiches + '/') : RAW_DEFAULT;
    const poster = base + 'BRK-' + s.cat + '.png';
    const links = [main.link].concat(s.group.map(function (g) { return g.link; })).slice(0, 3).join(' ');
    if (dry) { report += 'POST PRÉPARÉ :\n' + text + '\nAffiche : ' + poster + '\nSources : ' + links + '\n'; return; }
    if (log.length >= Number(p.getProperty('BRK_MAX') || 4)) return;
    if (log.length && now - log[log.length - 1] < Number(p.getProperty('BRK_GAP') || 45) * 60000) return;
    const t = new Date(now.getTime() + Number(p.getProperty('BRK_DELAY') || 10) * 60000);
    const id = 'BRK' + Utilities.formatDate(now, tz_(), 'yyyyMMddHHmm') + pg.key.slice(0, 4);
    const row = [id, Utilities.formatDate(t, tz_(), 'yyyy-MM-dd'), Utilities.formatDate(t, tz_(), 'HH:mm'), 'Facebook', 'Alerte actu', main.title, '', text, '', poster, links, 'Alerte', 'À valider', 'Mail envoyé le ' + stamp_(now), pg.nom];
    const sh = cal_(), rg = sh.getRange(sh.getLastRow() + 1, 1, 1, row.length);
    rg.setNumberFormat('@');
    rg.setValues([row]);
    sendApproval_(row, t, pg);
    log.push(now.getTime()); seen.push({ t: now.getTime(), sig: s.sig });
    p.setProperty('BRK_LOG_' + pg.key, JSON.stringify(log));
    p.setProperty('BRK_SEEN_' + pg.key, JSON.stringify(seen));
  });
  if (dry) say_('Veille actu : essai', report);
}

// Texte des posts "Jour J", ecrit a partir des flux, 3 heures avant.
function genJourJ_(r) {
  const label = String(r[C.PILIER] + ' ' + r[C.SUJET]).toLowerCase();
  const cat = label.indexOf('sport') >= 0 ? 'sport' : (label.indexOf('crypto') >= 0 ? 'crypto' : 'tech');
  const p = props_(), used = JSON.parse(p.getProperty('JJ_USED') || '[]');
  let items = [];
  FEEDS.filter(function (f) { return f.cat === cat && f.lang === 'fr'; }).forEach(function (f) { try { items = items.concat(feedItems_(f)); } catch (e) {} });
  items = items.filter(function (i) { return used.indexOf(i.link) < 0 && new Date() - i.date < 36 * 3600000; }).sort(function (a, b) { return b.date - a.date; });
  if (!items.length) return '';
  const it = items[0];
  used.push(it.link);
  p.setProperty('JJ_USED', JSON.stringify(used.slice(-60)));
  const chapo = cut_(noDash_(it.desc), 230);
  return ({ sport: 'LE POINT SPORT', crypto: 'LE POINT CRYPTO', tech: 'LE POINT NUMÉRIQUE' })[cat] + '\n\n' + noDash_(it.title) + '\n\n' + (chapo ? chapo + '\n\n' : '') + 'Source : ' + it.src + '.\n\nVotre avis ? Dites en commentaire.';
}

function setup() {
  const p = props_();
  if (!p.getProperty('SECRET')) p.setProperty('SECRET', Utilities.getUuid() + Utilities.getUuid());
  if (!p.getProperty('ADMIN_EMAIL')) p.setProperty('ADMIN_EMAIL', Session.getEffectiveUser().getEmail());
  if (!p.getProperty('PUBLISH_ENABLED')) p.setProperty('PUBLISH_ENABLED', 'false');
  ensureSheets_();
  ScriptApp.getProjectTriggers().forEach(function (t) { if (['tick', 'breaking_', 'tokenReminder'].indexOf(t.getHandlerFunction()) >= 0) ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('tick').timeBased().everyMinutes(5).create();
  ScriptApp.newTrigger('breaking_').timeBased().everyMinutes(10).create();
  ScriptApp.newTrigger('tokenReminder').timeBased().everyDays(1).atHour(7).create();
  say_('Agent installé', 'Les trois tâches automatiques sont actives. L\'agent est en mode simulation : il ne publie rien tant que tu n\'as pas choisi, dans le menu, Publier pour de vrai. Étapes suivantes : onglet Mode emploi.');

function menuSwitch() {
  const p = props_(), on = p.getProperty('PUBLISH_ENABLED') === 'true';

function menuTest() {
  const pg = selectedPage_();
  if (!pg) { say_('Test', 'Dans l\'onglet Pages, clique sur une ligne de page, puis relance ce test.'); return; }
  const out = ['Page : ' + pg.nom + ' (' + pg.reseau + ', ' + (pg.actif ? 'active' : 'inactive') + ')'];
  out.push(/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(pg.email) ? 'Email de validation : ' + pg.email : 'Email de validation manquant ou invalide.');
  out.push('Règle des posts rédigés : ' + (pg.regle === 'validation' ? 'validation obligatoire' : 'veto (publié sans réponse)'));
  try {
    if (pg.reseau === 'Facebook') {
      const tok = fbToken_(pg);
      const j = JSON.parse(UrlFetchApp.fetch('https://graph.facebook.com/' + pg.id + '?fields=id,name&access_token=' + encodeURIComponent(tok), { muteHttpExceptions: true }).getContentText());
      out.push(j.name ? 'Facebook : connexion réussie, Page « ' + j.name + ' ».' : 'Facebook : ' + JSON.stringify(j).slice(0, 200));
    } else {
      const exp = props_().getProperty('LI_EXP_' + pg.key);
      out.push(props_().getProperty('LI_TOKEN_' + pg.key) ? 'LinkedIn : connecté, jeton valable jusqu\'au ' + exp + '.' : 'LinkedIn : pas encore connecté. Menu, Envoyer le lien LinkedIn.');
    }
  } catch (e) { out.push('Problème : ' + e.message); }
  say_('Test de la page', out.join('\n'));
}
